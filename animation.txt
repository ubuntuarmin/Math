function fixMaterials(scene, exclude) {
scene.materials.forEach(material => {
material.backFaceCulling = false;
material.alphaMode = BABYLON.Constants.ALPHA_DISABLE;
material.transparencyMode = null;
material.needAlphaBlending = () => false;
material.needAlphaTesting = () => true;
material.usePhysicalLightFalloff = false;

const textures = [
material.diffuseTexture, material.albedoTexture, material.opacityTexture,
material.emissiveTexture, material.bumpTexture, material.normalTexture,
material.metallicTexture, material.roughnessTexture, material.reflectionTexture
].filter(Boolean);

textures.forEach(tex => {
if (!tex.name.includes("Ground")) {
tex.updateSamplingMode(BABYLON.Texture.NEAREST_NEAREST);
}
});
// PERFORMANCE OPTIMIZATION: Freeze materials once loaded
material.freeze();
});
}

async function createNavigationPluginAsync() {
const isFirefox = navigator.userAgent.includes("Firefox");
if (!isFirefox) {
return await ADDONS.CreateNavigationPluginAsync();
} else {
const recastVersion = "0.43.0";
const recastBase = "https://unpkg.com/@recast-navigation";
importShim.addImportMap({
imports: {
"@recast-navigation/core": `${recastBase}/core@${recastVersion}/dist/index.mjs`,
"@recast-navigation/wasm": `${recastBase}/wasm@${recastVersion}/dist/recast-navigation.wasm-compat.js`,
"@recast-navigation/generators": `${recastBase}/generators@${recastVersion}/dist/index.mjs`,
},
});
const [core, generators] = await Promise.all([
importShim(`${recastBase}/core@${recastVersion}/dist/index.mjs`),
importShim(`${recastBase}/generators@${recastVersion}/dist/index.mjs`),
]);
const recast = { ...core, ...generators };
await recast.init();
return await ADDONS.CreateNavigationPluginAsync({ instance: recast });
}
}

export const createScene = async function () {
engine.hideLoadingUI();
var _checkSupport = function() {
if (engine.getCaps().texelFetch && engine.getCaps().colorBufferFloat && engine.getCaps().blendFloat) { return true; }
const panel = BABYLON.GUI.AdvancedDynamicTexture.CreateFullscreenUI("UI");
const info = new BABYLON.GUI.TextBlock();
info.text = engine.isWebGPU ? "** This device does not meet requirements. **" : " ** WebGL on this device doesn't meet requirements. Try with WebGPU instead. **";
info.color = "red"; info.fontSize = "24px"; info.fontStyle = "bold";
panel.addControl(info);
return false;
};

var scene = new BABYLON.Scene(engine);
scene.clearColor.set(0,0,0,1.);

var camera = new BABYLON.ArcRotateCamera("camera1", 0, 0.2, 70, new BABYLON.Vector3(0, 1.5, 0), scene);
camera.lowerBetaLimit = 0.1;
camera.upperBetaLimit = 1.5;
camera.lowerRadiusLimit = 6;
camera.upperRadiusLimit = 70;

if (!_checkSupport()) { return scene; }

const navigationPlugin = await createNavigationPluginAsync();

var plane = BABYLON.MeshBuilder.CreateGround("defground", {width: 25, height: 25}, scene);
plane.position.z = -5; plane.visibility = 0;
await BABYLON.ImportMeshAsync("https://raw.githubusercontent.com/CedricGuillemet/dump/master/starryassets/ancient_ruins05k-png.glb", scene);

var staticMeshes = [];
var staticMeshesNav = [];
scene.meshes.forEach((mesh) => {
mesh.receiveShadows = true;
staticMeshes.push(mesh);
if (mesh.name.includes("defground") ||mesh.name.includes("Ground") || mesh.name.includes("Stones") || mesh.name.includes("Stone_Sculpture_Sculpture_0")) {
staticMeshesNav.push(mesh);
}
// PERFORMANCE OPTIMIZATION: Freeze static environment meshes
mesh.freezeWorldMatrix();
mesh.doNotSyncBoundingInfo = true;
});
scene.getMeshByName("Ground_Ground_0").scaling.set(2,1,2);

const source = "https://raw.githubusercontent.com/CedricGuillemet/dump/master/starryassets/env/";
const files = [`${source}px.png`, `${source}py.png`, `${source}pz.png`, `${source}nx.png`, `${source}ny.png`, `${source}nz.png`];
var envTexture = BABYLON.CubeTexture.CreateFromImages(files, scene);

const skydome = BABYLON.MeshBuilder.CreateBox("sky", { size: 1000, sideOrientation: BABYLON.Mesh.BACKSIDE }, scene);
skydome.rotation.y = Math.PI; skydome.isPickable = false;
const sky = new BABYLON.BackgroundMaterial("skyMaterial", scene);
sky.reflectionTexture = envTexture.clone();
sky.reflectionTexture.coordinatesMode = BABYLON.Texture.SKYBOX_MODE;
skydome.material = sky;

var light00 = new BABYLON.SpotLight("*spot00", new BABYLON.Vector3(0, 12, 8), new BABYLON.Vector3(0, -1, -1), 1.2, 20, scene);
light00.shadowMinZ = 1; light00.shadowMaxZ = 400; light00.intensity = 4; light00.angle = Math.PI * 0.5; light00.diffuse.set(1,0.5,0.1);

if (!engine.hostInformation.isMobile) {
var shadowGenerator = new BABYLON.ShadowGenerator(512, light00);
shadowGenerator.bias = 0.001; shadowGenerator.normalBias = 0.02;
shadowGenerator.usePoissonSampling = true; shadowGenerator.setDarkness(0.3);
staticMeshes.forEach((mesh)=>{ shadowGenerator.getShadowMap().renderList.push(mesh); });
}

const maxAgentRadius = 0.15; const AGENT_COUNT = 30; const cellSize = 0.5;
const navmeshParameters = {cs: cellSize, ch: 0.5, walkableRadius: Math.ceil(maxAgentRadius / cellSize), maxObstacles: 0 };
await navigationPlugin.createNavMeshAsync(staticMeshesNav, navmeshParameters);

fixMaterials(scene, []);
const crowd = navigationPlugin.createCrowd(AGENT_COUNT, maxAgentRadius, scene);
const agentParams = { radius: 0.5, height: 0.5, maxAcceleration: 4.0, maxSpeed: 2.0, separationWeight: 1.0 };
const clustered = new BABYLON.ClusteredLightContainer("clustered", [], scene);
var agentDatas = [];

var getRandomPos = function() {
var pickedPoint = new BABYLON.Vector3(Math.random() * 20 - 10, 1, Math.random() * 20 - 10);
return navigationPlugin.getClosestPoint(pickedPoint);
};

var randomGoto = function(index) {
const agents = crowd.getAgents();
const destination = getRandomPos();
crowd.agentGoto(agents[index], destination);
return destination;
};

// PERFORMANCE OPTIMIZATION: Drastically reduce the number of dynamic point lights
const MAX_LIGHTS = 6;

BABYLON.LoadAssetContainerAsync("https://raw.githubusercontent.com/CedricGuillemet/dump/master/starryassets/lowpoly_firefly-lowres-texture.glb", scene).then((container) => {
container.meshes[0].scaling.scaleInPlace(2);

for (let i = 0; i < AGENT_COUNT ; i++) {
const plane2Entries = container.instantiateModelsToScene(undefined, false, { doNotInstantiate: true });
const firefly = plane2Entries.rootNodes[0]; firefly.position.y = 1.;
plane2Entries.animationGroups[1].start(true, Math.random() * 0.1 + 0.9);
let pointLight = null;
if (i < MAX_LIGHTS) {
pointLight = new BABYLON.PointLight("point" + i, new BABYLON.Vector3(0,0.1,0), scene);
pointLight.diffuse = new BABYLON.Color3(1,0.5,0.1); pointLight.range = 2; pointLight.intensity = 8;
clustered.addLight(pointLight); pointLight.parent = firefly;
}

const agentTransform = new BABYLON.TransformNode("agentTransform", scene);
const agentIndex = crowd.addAgent(getRandomPos(), agentParams, agentTransform);
firefly.parent = agentTransform;

agentDatas[i] = {mesh: agentTransform, idx: agentIndex, target: randomGoto(i), light: pointLight, ttc: 0};
}
container.meshes[0].setEnabled(false);
});

const guiTexture = BABYLON.GUI.AdvancedDynamicTexture.CreateFullscreenUI("LoadingScreenHUD");
const overlay = new BABYLON.GUI.Rectangle();
overlay.width = "100%";
overlay.height = "100%";
overlay.background = "rgba(0, 0, 0, 0.75)";
overlay.thickness = 0;
overlay.alpha = 0;
guiTexture.addControl(overlay);

const mainContainer = new BABYLON.GUI.StackPanel();
mainContainer.horizontalAlignment = BABYLON.GUI.Control.HORIZONTAL_ALIGNMENT_CENTER;
mainContainer.verticalAlignment = BABYLON.GUI.Control.VERTICAL_ALIGNMENT_CENTER;
overlay.addControl(mainContainer);

const titleText = new BABYLON.GUI.TextBlock();
titleText.text = "Math-Katy";
titleText.color = "white";
titleText.fontSize = "80px";
titleText.fontFamily = "Arial";
titleText.fontWeight = "bold";
titleText.outlineColor = "black";
titleText.outlineWidth = 8;
titleText.height = "100px";
mainContainer.addControl(titleText);

const subtitleText = new BABYLON.GUI.TextBlock();
subtitleText.text = "an open source community";
subtitleText.color = "#d3d3d3";
subtitleText.fontSize = "26px";
subtitleText.fontFamily = "Arial";
subtitleText.fontStyle = "italic";
subtitleText.outlineColor = "black";
subtitleText.outlineWidth = 5;
subtitleText.height = "40px";
mainContainer.addControl(subtitleText);

const cameraAnimDuration = 12000;
const titleFadeInStart = 9000;
const holdTitleDuration = 16000;
const endSequenceTime = 18000;

let elapsedTime = 0;
let sequenceFinished = false;

const startRadius = 70, endRadius = 10;
const startBeta = 0.2, endBeta = 1.35;
const totalSpins = Math.PI * 2.5;

// PERFORMANCE OPTIMIZATION: Combined everything into a single render loop
const renderObserver = scene.onBeforeRenderObservable.add(() => {
let dt = engine.getDeltaTime();
let dtSec = dt * 0.001;
elapsedTime += dt;

// --- 1. FIREFLY AGENT UPDATES ---
for(let i = 0; i < agentDatas.length; i++) {
var ag = agentDatas[i];
if (!ag || !ag.mesh) continue;
ag.ttc -= dt;
ag.mesh.position = crowd.getAgentPosition(ag.idx);
let vel = crowd.getAgentVelocity(ag.idx);
crowd.getAgentNextTargetPathToRef(ag.idx, ag.target);
if (vel.length() > 0.01) {
vel.normalize();
ag.mesh.rotation.y += (Math.atan2(vel.x, vel.z) - ag.mesh.rotation.y) * 0.05;
}
if (BABYLON.Vector3.Distance(ag.mesh.position, ag.target) < 0.1 || ag.ttc < 0) {
ag.target = randomGoto(i); ag.ttc = Math.random() * 2000 + 10000;
}
}

// --- 2. CINEMATIC SEQUENCE ---
if (sequenceFinished) return;

if (elapsedTime <= cameraAnimDuration) {
let progress = elapsedTime / cameraAnimDuration;
let ease = 1 - Math.pow(1 - progress, 4);

camera.radius = startRadius - ((startRadius - endRadius) * ease);
camera.beta = startBeta + ((endBeta - startBeta) * ease);
camera.alpha = totalSpins * ease;
} else {
camera.alpha += 0.0001 * dt;
}

if (elapsedTime > titleFadeInStart && elapsedTime <= holdTitleDuration) {
let fadeProgress = (elapsedTime - titleFadeInStart) / (cameraAnimDuration - titleFadeInStart);
overlay.alpha = Math.min(1, fadeProgress);
}

if (elapsedTime > cameraAnimDuration && elapsedTime <= holdTitleDuration) {
let scale = 1 + Math.sin(elapsedTime * 0.002) * 0.02;
mainContainer.scaleX = scale;
mainContainer.scaleY = scale;
}

// --- 3. FADE TO BLACK ---
if (elapsedTime > holdTitleDuration) {
let fadeToBlackProgress = (elapsedTime - holdTitleDuration) / (endSequenceTime - holdTitleDuration);
fadeToBlackProgress = Math.min(1, Math.max(0, fadeToBlackProgress));
// Background goes to solid black (alpha 1)
overlay.background = `rgba(0, 0, 0, ${0.75 + (0.25 * fadeToBlackProgress)})`;
// Text fades out
mainContainer.alpha = 1 - fadeToBlackProgress;

if (elapsedTime >= endSequenceTime) {
sequenceFinished = true;
// At this point, the screen is completely black.
}
}
});
return scene;
}
